# PostgreSQL Template Database Integration Tests Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the integration suite's single destructive PostgreSQL database with one migrated template and one ephemeral clone/shared pool per package so packages can run concurrently.

**Architecture:** A new `internal/testdb` harness validates the base test DSN, serializes template migration and cloning with a connection-bound PostgreSQL advisory lock, and runs each package against a uniquely named clone. Integration packages bind the clone pool in `TestMain`; existing tests retain their within-package reset and fixture behavior while Make and CI remove `-p 1`.

**Tech Stack:** Go 1.27, `database/sql`, pgx stdlib driver, PostgreSQL 17 template databases, testify, GNU Make, GitHub Actions

**Spec:** `docs/superpowers/specs/2026-09-15-postgres-template-integration-tests-design.md`

## Global Constraints

- `TEST_DATABASE_URL` is mandatory and its database name must end exactly in `_test`.
- Production `internal/database.Open` and migration behavior must remain unchanged.
- Integration CI must retain `-race` on every pull request.
- This phase creates one clone and one shared pool per package process, not per test.
- Do not add integration `t.Parallel()` or change current fixture/reset semantics.
- Clone identifiers must contain only lowercase ASCII letters, digits, and underscores and must not exceed PostgreSQL's 63-byte identifier limit.
- SQL identifiers must be generated by the harness, validated, and quoted; never interpolate user-controlled identifiers.
- Never include passwords or complete DSNs in logs or returned errors.
- Normal teardown must never drop the base `_test` database or the persistent `_test_template` database.
- Do not add a new dependency; pgx, `lib/pq`, and testify are already available.
- Commit steps below are conditional: execute them only if the user explicitly requests commits.

Create `internal/testdb/config.go` with these definitions and rules:

```go
package testdb

import (
	"crypto/rand"
	"encoding/hex"
	"errors"
	"fmt"
	"net/url"
	"regexp"
	"strings"
)

var allowedPackages = map[string]struct{}{
	"auth": {}, "catalog": {}, "database": {},
	"sales": {}, "shift": {}, "tables": {},
}

var (
	identifierPattern = regexp.MustCompile(`^[a-z][a-z0-9_]*$`)
	suffixPattern     = regexp.MustCompile(`^[0-9a-f]{12}$`)
)

type config struct {
	baseConfig
	cloneName      string
	cloneDSN       string
}

type baseConfig struct {
	baseURL       *url.URL
	baseName      string
	templateName  string
	maintenanceDSN string
	templateDSN   string
}

func parseBaseConfig(rawURL string) (baseConfig, error) {
	if rawURL == "" {
		return baseConfig{}, errors.New("TEST_DATABASE_URL is required")
	}
	parsed, err := url.Parse(rawURL)
	if err != nil {
		return baseConfig{}, errors.New("TEST_DATABASE_URL is not a valid postgres URL")
	}
	if parsed.Scheme != "postgres" && parsed.Scheme != "postgresql" {
		return baseConfig{}, errors.New("TEST_DATABASE_URL must be a postgres URL")
	}
	if parsed.Host == "" {
		return baseConfig{}, errors.New("TEST_DATABASE_URL host is required")
	}
	escapedName := strings.TrimPrefix(parsed.EscapedPath(), "/")
	if escapedName == "" || strings.Contains(escapedName, "/") {
		return baseConfig{}, errors.New("TEST_DATABASE_URL must contain one database name")
	}
	baseName, err := url.PathUnescape(escapedName)
	if err != nil {
		return baseConfig{}, errors.New("TEST_DATABASE_URL contains an invalid database name")
	}
	if !identifierPattern.MatchString(baseName) {
		return baseConfig{}, errors.New("TEST_DATABASE_URL database name is not a safe test identifier")
	}
	if !strings.HasSuffix(baseName, "_test") {
		return baseConfig{}, errors.New("TEST_DATABASE_URL database name must end in _test")
	}
	templateName := baseName + "_template"
	if len(templateName) > 63 {
		return baseConfig{}, errors.New("template database name exceeds 63 bytes")
	}
	return baseConfig{
		baseURL:        parsed,
		baseName:       baseName,
		templateName:   templateName,
		maintenanceDSN: databaseDSN(parsed, "postgres"),
		templateDSN:    databaseDSN(parsed, templateName),
	}, nil
}

func parseConfig(rawURL, packageName, suffix string) (config, error) {
	base, err := parseBaseConfig(rawURL)
	if err != nil {
		return config{}, err
	}
	if _, ok := allowedPackages[packageName]; !ok {
		return config{}, fmt.Errorf("unsupported integration package %q", packageName)
	}
	if !suffixPattern.MatchString(suffix) {
		return config{}, errors.New("invalid clone suffix")
	}
	cloneName := base.baseName + "_" + packageName + "_" + suffix
	if len(cloneName) > 63 {
		return config{}, errors.New("clone database name exceeds 63 bytes")
	}
	if !identifierPattern.MatchString(cloneName) {
		return config{}, errors.New("clone database name is not a safe identifier")
	}
	return config{
		baseConfig: base,
		cloneName:  cloneName,
		cloneDSN:   databaseDSN(base.baseURL, cloneName),
	}, nil
}

func databaseDSN(base *url.URL, databaseName string) string {
	copy := *base
	copy.Path = "/" + databaseName
	copy.RawPath = ""
	return copy.String()
}

func newSuffix() (string, error) {
	var raw [6]byte
	if _, err := rand.Read(raw[:]); err != nil {
		return "", fmt.Errorf("generate clone suffix: %w", err)
	}
	return hex.EncodeToString(raw[:]), nil
}

func isEphemeralClone(baseName, databaseName string) bool {
	for packageName := range allowedPackages {
		prefix := baseName + "_" + packageName + "_"
		if strings.HasPrefix(databaseName, prefix) && suffixPattern.MatchString(strings.TrimPrefix(databaseName, prefix)) {
			return true
		}
	}
	return false
}

func sanitizeError(err error, rawURL string) error {
	if err == nil {
		return nil
	}
	message := err.Error()
	parsed, parseErr := url.Parse(rawURL)
	if parseErr != nil {
		return errors.New("integration database operation failed")
	}
	message = strings.ReplaceAll(message, rawURL, parsed.Redacted())
	if parsed.User != nil {
		if password, ok := parsed.User.Password(); ok && password != "" {
			for _, value := range []string{password, url.QueryEscape(password), url.PathEscape(password)} {
				message = strings.ReplaceAll(message, value, "xxxxx")
			}
		}
	}
	return errors.New(message)
}
```

Add `errors` to the imports. Do not include `rawURL` in validation errors. Use `url.URL.Redacted()` only if a URL must appear in diagnostics.

```go
package testdb

import (
	"context"
	"database/sql"
	"errors"
	"fmt"
	"os"
	"testing"
	"time"

	"github.com/Mirai3103/pos-cafe/internal/database"
	"github.com/lib/pq"
)

const templateLockID int64 = 7142982

type instance struct {
	db          *sql.DB
	maintenance *sql.DB
	cfg         config
}

func Run(m *testing.M, packageName string, bind func(*sql.DB)) int {
	rawURL := os.Getenv("TEST_DATABASE_URL")
	suffix, err := newSuffix()
	if err != nil {
		fmt.Fprintf(os.Stderr, "integration database setup for %s: %v\n", packageName, sanitizeError(err, rawURL))
		return 1
	}
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	inst, err := provision(ctx, rawURL, packageName, suffix)
	cancel()
	if err != nil {
		fmt.Fprintf(os.Stderr, "integration database setup for %s: %v\n", packageName, sanitizeError(err, rawURL))
		return 1
	}
	bind(inst.db)
	testCode := m.Run()
	cleanupCtx, cleanupCancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cleanupCancel()
	cleanupErr := inst.Close(cleanupCtx)
	if cleanupErr != nil {
		fmt.Fprintf(os.Stderr, "integration database cleanup for %s: %v\n", packageName, sanitizeError(cleanupErr, rawURL))
	}
	return finalExitCode(testCode, cleanupErr)
}

func finalExitCode(testCode int, cleanupErr error) int {
	if testCode != 0 {
		return testCode
	}
	if cleanupErr != nil {
		return 1
	}
	return 0
}
```

Implement `provision` in this exact order:

1. Call `parseConfig` before opening any connection.
2. Open and ping `cfg.maintenanceDSN` with driver `pgx`.
3. Reserve `maintenance.Conn(ctx)` and keep that `*sql.Conn` until the advisory lock is released.
4. Execute `SELECT pg_advisory_lock($1)` on that connection.
5. Query `pg_database` for `cfg.templateName`; if absent, execute `CREATE DATABASE <quoted-template> TEMPLATE template0` on the same connection.
6. Open `cfg.templateDSN` with `database.Open`, apply pending migrations, and close that pool.
7. Execute `CREATE DATABASE <quoted-clone> TEMPLATE <quoted-template>` on the reserved maintenance connection. If the random name already exists, fail setup; never drop a clone this invocation did not create.
8. Execute `SELECT pg_advisory_unlock($1)` on the same connection and close the reserved connection.
9. Open `cfg.cloneDSN` using `sql.Open("pgx", cfg.cloneDSN)`, configure 10 max-open and 5 max-idle connections with a five-minute lifetime and one-minute idle timeout, then ping.
10. Return an `instance`; track whether this invocation successfully created the clone and only attempt to drop it on later setup failure when that flag is true.

Quote generated identifiers with `pq.QuoteIdentifier` only after `parseConfig` validates them. Join teardown errors with `errors.Join`. `instance.Close` must close the package pool before issuing `DROP DATABASE <clone> WITH (FORCE)`, then close the maintenance pool.

Factor lock ownership into `withTemplateLock(ctx context.Context, maintenance *sql.DB, fn func(*sql.Conn) error) (err error)`. It reserves one `*sql.Conn`, acquires the lock, invokes `fn` on that same connection, and in deferred cleanup unlocks on that connection before closing it. Join callback, unlock, and connection-close errors rather than replacing the first error.

Implement `Cleanup` by calling `parseBaseConfig(rawURL)`, opening its maintenance DSN, and using this flow:

```go
rows, err := maintenance.QueryContext(ctx, `SELECT datname FROM pg_database`)
// Collect names, close rows, and check rows.Err before issuing more queries.
// For each collected name:
// 1. Continue unless isEphemeralClone(base.baseName, name) is true.
// 2. Query SELECT count(*) FROM pg_stat_activity WHERE datname = $1.
// 3. Continue when the count is non-zero.
// 4. Execute DROP DATABASE <pq.QuoteIdentifier(name)> WITH (FORCE).
```

Close the maintenance pool on every return and sanitize returned connection/driver errors with `sanitizeError`. Collect independent drop failures with `errors.Join` so one stale clone does not prevent cleanup of later candidates. Do not drop the base or template under any condition.

```go
package testdb

import (
	"context"
	"os"
	"testing"

	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
)

func TestProvisionCreatesMigratedIsolatedClones(t *testing.T) {
	rawURL := os.Getenv("TEST_DATABASE_URL")
	require.NotEmpty(t, rawURL)
	ctx := context.Background()

	firstSuffix, err := newSuffix()
	require.NoError(t, err)
	first, err := provision(ctx, rawURL, "sales", firstSuffix)
	require.NoError(t, err)
	t.Cleanup(func() { require.NoError(t, first.Close(context.Background())) })

	secondSuffix, err := newSuffix()
	require.NoError(t, err)
	second, err := provision(ctx, rawURL, "sales", secondSuffix)
	require.NoError(t, err)
	t.Cleanup(func() { require.NoError(t, second.Close(context.Background())) })

	var migrationCount int
	require.NoError(t, first.db.QueryRow(`SELECT count(*) FROM schema_migrations`).Scan(&migrationCount))
	assert.Positive(t, migrationCount)

	_, err = first.db.Exec(`CREATE TABLE clone_isolation_probe (id integer PRIMARY KEY)`)
	require.NoError(t, err)
	_, err = first.db.Exec(`INSERT INTO clone_isolation_probe (id) VALUES (1)`)
	require.NoError(t, err)

	var exists bool
	require.NoError(t, second.db.QueryRow(`SELECT to_regclass('public.clone_isolation_probe') IS NOT NULL`).Scan(&exists))
	assert.False(t, exists)
}
```
